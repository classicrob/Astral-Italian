cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(PetariDrawObserverCore LANGUAGES CXX)
    enable_testing()
    set(BUILD_TESTING ON)
endif()
add_library(petari_draw_observer_core STATIC draw_observer.cpp draw_observer_json.cpp)
target_compile_features(petari_draw_observer_core PUBLIC cxx_std_17)
target_include_directories(petari_draw_observer_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(BUILD_TESTING)
    add_executable(petari_draw_observer_controls draw_observer_tests.cpp)
    target_link_libraries(petari_draw_observer_controls PRIVATE petari_draw_observer_core)
    target_compile_options(petari_draw_observer_controls PRIVATE -UNDEBUG)
    add_test(NAME native_draw_observer_core COMMAND petari_draw_observer_controls)
endif()
if(TARGET petari_native_config)
    add_library(petari_draw_observer_game STATIC draw_snapshot.cpp)
    target_link_libraries(petari_draw_observer_game PUBLIC petari_native_config petari_draw_observer_core)
endif()

# Explicit future integration call; defining this function changes no existing target.
function(petari_link_draw_observer_to_game)
    if(NOT TARGET petari_game_objects)
        message(FATAL_ERROR "Draw observer linkage requires the native game object target")
    endif()
    target_link_libraries(petari_game_objects PUBLIC petari_draw_observer_core)
endfunction()

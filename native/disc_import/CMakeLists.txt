cmake_minimum_required(VERSION 3.25)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(PetariDiscImport LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    include(CTest)
endif()
get_filename_component(_petari_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${_petari_root}/native/cmake/AuroraSource.cmake")
include(FetchContent)
FetchContent_GetProperties(aurora)
if(NOT aurora_POPULATED)
    # Only use Aurora's dependency modules; do not configure its renderer or SDL.
    FetchContent_Declare(aurora
        GIT_REPOSITORY https://github.com/encounter/aurora.git
        GIT_TAG ${PETARI_AURORA_REVISION}
        SOURCE_SUBDIR petari-disc-import-no-subproject)
    FetchContent_MakeAvailable(aurora)
endif()
include("${aurora_SOURCE_DIR}/cmake/AuroraDependencyVersions.cmake")
set(AURORA_NOD_PROVIDER auto CACHE STRING "nod dependency provider")
set(AURORA_NOD_LINKAGE static)
include("${aurora_SOURCE_DIR}/cmake/AuroraNodProvider.cmake")
if(NOT TARGET nod::nod_static)
    message(FATAL_ERROR "The disc importer requires static nod; check Aurora's nod provider/package")
endif()

add_library(petari_disc_import STATIC disc_import.cpp)
target_compile_features(petari_disc_import PUBLIC cxx_std_17)
target_include_directories(petari_disc_import PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
# Share the exact FST parser used at mount, without linking the SDK/drive service.
if(NOT TARGET petari_dvd_fst)
    add_library(petari_dvd_fst STATIC ../platform/dvd/dvd_fst.cpp)
    target_include_directories(petari_dvd_fst PUBLIC ../platform/include ../platform/dvd)
    target_compile_features(petari_dvd_fst PUBLIC cxx_std_17)
endif()
target_link_libraries(petari_disc_import PRIVATE petari_dvd_fst nod::nod_static)
target_compile_options(petari_disc_import PRIVATE -Wno-deprecated-declarations)
add_executable(petari_disc_import_cli main.cpp)
set_target_properties(petari_disc_import_cli PROPERTIES OUTPUT_NAME petari_disc_import)
target_link_libraries(petari_disc_import_cli PRIVATE petari_disc_import)
if(BUILD_TESTING)
    add_executable(petari_disc_import_tests ../tests/disc_import_tests.cpp)
    target_include_directories(petari_disc_import_tests PRIVATE ../platform/include ../platform/dvd)
    target_compile_options(petari_disc_import_tests PRIVATE -Wno-deprecated-declarations)
    target_link_libraries(petari_disc_import_tests PRIVATE petari_disc_import)
    add_test(NAME native_disc_import COMMAND petari_disc_import_tests)
endif()

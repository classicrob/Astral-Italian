cmake_minimum_required(VERSION 3.25)
project(PadProbe LANGUAGES C OBJC)
if(NOT APPLE)
    message(FATAL_ERROR "PadProbe uses macOS IOHID and dyld diagnostics")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Target architecture" FORCE)
endif()
set(PADPROBE_SDL_DIR "" CACHE PATH "Optional scratch override: root containing src/sdl/include and build/sdl/libSDL3.a")
set(PADPROBE_LIBUSB_DIR "" CACHE PATH "Optional scratch override: libusb prefix (requires PADPROBE_SDL_DIR too)")
set(PADPROBE_GIT_ID "" CACHE STRING "Probe source revision, defaults to git describe --always --dirty")
if(NOT PADPROBE_GIT_ID)
    execute_process(COMMAND git describe --always --dirty WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        OUTPUT_VARIABLE PADPROBE_GIT_ID OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
get_filename_component(_native "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(PADPROBE_SDL_DIR OR PADPROBE_LIBUSB_DIR)
    if(NOT EXISTS "${PADPROBE_SDL_DIR}/build/sdl/libSDL3.a" OR
       NOT EXISTS "${PADPROBE_SDL_DIR}/src/sdl/include/SDL3/SDL.h" OR
       NOT EXISTS "${PADPROBE_LIBUSB_DIR}/include/libusb-1.0/libusb.h")
        message(FATAL_ERROR "Scratch override requires both PADPROBE_SDL_DIR and PADPROBE_LIBUSB_DIR")
    endif()
    set(_sdl_target "${PADPROBE_SDL_DIR}/build/sdl/libSDL3.a")
    set(_sdl_include "${PADPROBE_SDL_DIR}/src/sdl/include")
    set(_sdl_license "${PADPROBE_SDL_DIR}/src/sdl/LICENSE.txt")
    set(_usb_prefix "${PADPROBE_LIBUSB_DIR}")
    file(WRITE "${CMAKE_BINARY_DIR}/dependency-paths.tsv" "SCRATCH OVERRIDE: ${PADPROBE_SDL_DIR}\nlibusb: ${_usb_prefix}\n")
else()
    # Same recipe as native/tools/libusb_load_check: populate Aurora only for
    # SDL patch scripts. No GX, assets, Dawn, or game targets are configured.
    include(FetchContent)
    FetchContent_Declare(aurora
        URL https://github.com/encounter/aurora/archive/08122911e8621acb7ded6563813b264bec1494b5.tar.gz
        DOWNLOAD_EXTRACT_TIMESTAMP FALSE)
    FetchContent_GetProperties(aurora)
    if(NOT aurora_POPULATED)
        FetchContent_Populate(aurora)
    endif()
    include("${_native}/cmake/PadDependencies.cmake")
    FetchContent_MakeAvailable(SDL)
    include("${aurora_SOURCE_DIR}/cmake/AuroraDependencyVersions.cmake")
    include("${_native}/cmake/GraphicsDependencies.cmake")
    file(WRITE "${CMAKE_BINARY_DIR}/dependency-paths.tsv" "")
    petari_record_pad_dependencies("${CMAKE_BINARY_DIR}/dependency-paths.tsv")
    set(_sdl_target SDL3::SDL3-static)
    set(_sdl_include "${sdl_SOURCE_DIR}/include")
    set(_sdl_license "${sdl_SOURCE_DIR}/LICENSE.txt")
    set(_usb_prefix "${PETARI_LIBUSB_PREFIX}")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/padprobe-libusb-prefix.txt" "${_usb_prefix}\n")
configure_file("${_sdl_license}" "${CMAKE_BINARY_DIR}/padprobe-SDL-LICENSE.txt" COPYONLY)
add_library(padprobe_core STATIC probe_core.c)
target_include_directories(padprobe_core PUBLIC "${_sdl_include}")
target_compile_options(padprobe_core PRIVATE -Wall -Wextra -Werror)
add_executable(padprobe padprobe.c)
target_include_directories(padprobe PRIVATE "${_usb_prefix}/include/libusb-1.0")
target_compile_definitions(padprobe PRIVATE PADPROBE_GIT_ID="${PADPROBE_GIT_ID}")
target_compile_options(padprobe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(padprobe PRIVATE padprobe_core "${_sdl_target}" iconv)
foreach(framework Cocoa IOKit CoreFoundation CoreAudio AudioToolbox CoreHaptics GameController
        ForceFeedback Metal QuartzCore CoreVideo CoreServices Carbon AVFoundation CoreMedia
        UniformTypeIdentifiers CoreGraphics ImageIO Foundation AppKit OpenGL)
    target_link_libraries(padprobe PRIVATE "-framework ${framework}")
endforeach()
# SDL currently loads the plain libusb name; this also supports that scratch
# revision. Sol A's first-try ../Frameworks path uses the identical package layout.
set_target_properties(padprobe PROPERTIES BUILD_RPATH "@executable_path/../Frameworks"
    INSTALL_RPATH "@executable_path/../Frameworks")
add_executable(padprobe_selftest selftest.c)
target_link_libraries(padprobe_selftest PRIVATE padprobe_core)
target_compile_options(padprobe_selftest PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME padprobe_selftest COMMAND padprobe_selftest)
